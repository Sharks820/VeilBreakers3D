```csharp
    VisualElement veilPulseFlash,
    Label heroNameLabel,
    Label heroTitleLabel,
    VolumeProfileTransitioner volumeTransitioner,
    VeilDissolveController dissolveController,

    Action applyMusicParameters,
    Action applyLightingTransition,
    Action<HeroThemeConfig> applyStatBarCascade
)
```

Timeline per CONTEXT.md (locked decision):
```
t=0ms:   Info panel fades out (opacity 0 over 200ms) + music crossfade starts
t=100ms: Veil pulse flash (full-screen element, hero accent color, 0.1 opacity, 200ms fade-out)
t=200ms: 3D model dissolve OUT begins (400ms) + post-process lerp starts (0.8s) + old particles fade
t=400ms: New hero materializes (dissolve IN, 400ms) + new particles begin spawning
t=600ms: Info panel fades in with new hero data (opacity 1 over 300ms) + overlays shift
t=800ms: Stat bars cascade fill (HP -> ATK -> DEF -> SPD, 100ms apart)
t=800ms: Hero name/title glitch text reveal (~250ms)
```

Implementation using PrimeTween Sequence.Create().Insert(atTime):
```csharp
return Sequence.Create()
    // t=0ms: Info panel fade out + music
    .Insert(0f, Tween.StyleOpacity(infoPanel, 0f, 0.2f, Ease.InQuad))
    .InsertCallback(0f, () => applyMusicParameters())

    // t=100ms: Veil pulse flash
    .Insert(0.1f, BuildVeilPulseFlash(veilPulseFlash, newTheme.primaryColor))

    // t=200ms: Dissolve out + post-process
    .Insert(0.2f, dissolveController.AnimateDissolveOut(newTheme.dissolveDuration))
    .InsertCallback(0.2f, () => volumeTransitioner.LerpTo(newTheme, 0.8f))

    // t=400ms: Materialize
    .InsertCallback(0.4f, () => dissolveController.SetHeroProperties(newTheme))
    .Insert(0.4f, dissolveController.AnimateDissolveIn(newTheme.dissolveDuration))

    // t=600ms: Info panel fade in + overlays
    .Insert(0.6f, Tween.StyleOpacity(infoPanel, 1f, 0.3f, Ease.OutQuad))
    .InsertCallback(0.6f, () => overlayController.TransitionTo(newTheme, 0.5f))
    .InsertCallback(0.6f, () => applyLightingTransition())

    // t=800ms: Stat cascade + glitch text
    .InsertCallback(0.8f, () => applyStatBarCascade(newTheme))
    .Insert(0.8f, GlitchTextEffect.BuildGlitchReveal(heroNameLabel, heroNameLabel.text, newTheme.glitchResolveSpeed))
    .Insert(0.8f, GlitchTextEffect.BuildGlitchReveal(heroTitleLabel, heroTitleLabel.text, newTheme.glitchResolveSpeed));
```

BuildVeilPulseFlash helper: Creates a brief full-screen flash element animation:
- Set veilPulseFlash background-color to hero accent color
- Animate opacity from 0.1 to 0 over 200ms with Ease.OutQuad

**A. Update HeroStageController.cs -- Per-Hero Lighting Lerp + Nyx Rim Flicker**

Add new fields:

```csharp
private float _srcFillIntensity, _dstFillIntensity;

private float _srcRimIntensity, _dstRimIntensity;
```

Add new public method:
```csharp
/// <summary>
/// Lerps stage lighting from current values to the hero theme's lighting config over 0.6s.
/// Starts/stops Nyx rim flicker based on theme.rimFlicker.
/// </summary>
public void TransitionLighting(HeroThemeConfig theme, float duration = 0.6f)
{
    _lightLerpTween.Stop();

    // Cache source from current light values
    _srcFillColor = _fillLight.color;
    _srcFillIntensity = _fillLight.intensity;
    _srcRimColor = _rimLight.color;
    _srcRimIntensity = _rimLight.intensity;

    // Cache destination from theme
    _dstFillColor = theme.fillLightColor;
    _dstFillIntensity = theme.fillLightIntensity;
    _dstRimColor = theme.rimLightColor;
    _dstRimIntensity = theme.rimLightIntensity;

    _lightLerpTween = Tween.Custom(this, 0f, 1f, duration, Ease.InOutSine,
        (ctrl, t) => ctrl.ApplyLightLerp(t));

    // Rim flicker for Nyx
    if (theme.rimFlicker && !_isRimFlickering)
        StartRimFlicker();
    else if (!theme.rimFlicker && _isRimFlickering)
        StopRimFlicker();
}
```

StartRimFlicker(): Creates a looping PrimeTween that randomly varies `_rimLight.intensity` between 0.8x and 1.5x of the target intensity every 0.1-0.3s. Use `Tween.Custom` with `CycleMode.Restart` and `cycles: -1` for infinite loop. The variation should feel like unstable electrical energy, not a smooth sine wave -- use short duration (0.1-0.3s) tweens with different easing each cycle.

StopRimFlicker(): Stops the flicker tween, lerps rim light back to stable intensity.

ApplyLightLerp(float t):
- `_fillLight.color = Color.Lerp(_srcFillColor, _dstFillColor, t);`
- `_fillLight.intensity = Mathf.Lerp(_srcFillIntensity, _dstFillIntensity, t);`
- Same for rim light

Also add to the existing lighting rig: if the stage doesn't have an ambient light, add one. The theme has `ambientColor` that should tint the environment. Use `RenderSettings.ambientLight = Color.Lerp(srcAmbient, theme.ambientColor, t)` in the lerp or add a dedicated ambient light.

```csharp
/// <summary>
/// Updates parallax offsets based on mouse/stick position.
/// Call from MonoBehaviour.Update() via HeroThemeTransitioner.
/// Per CONTEXT.md: "UI panels shift 3-5px based on mouse position,
/// overlay layers at different rates for depth."
/// </summary>
public void UpdateParallax()
{
    // Read mouse position from InputManager (same pattern as ParallaxBackground.cs)
    Vector2 mousePos = InputManager.HasInstance
        ? InputManager.Instance.MousePosition
        : Vector2.zero;

    // Normalize to -1..+1 range from screen center
    Vector2 normalized = new Vector2(
        (mousePos.x - _screenCenter.x) / _screenCenter.x,
        (mousePos.y - _screenCenter.y) / _screenCenter.y
    );

    _targetParallaxOffset = normalized;

    // Smooth interpolation (same approach as ParallaxBackground)
    _currentParallaxOffset = Vector2.Lerp(
        _currentParallaxOffset,
        _targetParallaxOffset,
        Time.unscaledDeltaTime * kParallaxSmoothing
    );

    // Dirty check: skip style writes if offset hasn't changed meaningfully
    if (Mathf.Abs(_currentParallaxOffset.x - _previousParallaxOffset.x) < kParallaxDirtyThreshold &&
        Mathf.Abs(_currentParallaxOffset.y - _previousParallaxOffset.y) < kParallaxDirtyThreshold)
        return;
    _previousParallaxOffset = _currentParallaxOffset;

    // Apply per-layer parallax offsets (deepest moves least)
    ApplyTranslateOffset(_veilGlow, kVeilGlowParallax);     // 0.5px
    ApplyTranslateOffset(_scanlines, kScanlineParallax);     // 1.0px
    ApplyTranslateOffset(_vignette, kVignetteParallax);      // 2.0px

    // UI panels: 3-5px movement (per CONTEXT.md locked decision)
    ApplyTranslateOffset(_leftPanel, kPanelParallax);
    ApplyTranslateOffset(_rightPanel, kPanelParallax);
    ApplyTranslateOffset(_carousel, kPanelParallax * 0.8f);  // Carousel slightly less
}

private void ApplyTranslateOffset(VisualElement element, float multiplier)
{
    if (element == null) return;
    float x = _currentParallaxOffset.x * multiplier;
    float y = _currentParallaxOffset.y * multiplier;
    element.style.translate = new Translate(x, y);
}
```
